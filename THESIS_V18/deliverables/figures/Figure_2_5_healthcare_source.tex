% Healthcare and assistive care: monitoring of a patient with wearable and ambient sensors.
\begin{document}
\begin{tikzpicture}[
  font=\small,
  sen/.style={draw, thick, rounded corners=2pt, fill=gray!12, align=flush left, text width=36mm, inner sep=4pt},
  loc/.style={draw, thick, rounded corners=2pt, fill=blue!7, align=flush left, text width=44mm, inner sep=4pt},
  srv/.style={draw, thick, rounded corners=2pt, fill=green!12, align=flush left, text width=38mm, inner sep=4pt},
  res/.style={draw, thick, rounded corners=2pt, fill=orange!14, align=flush left, text width=44mm, inner sep=4pt},
  grp/.style={draw, thick, dashed, rounded corners=4pt, inner sep=6pt},
  arr/.style={-{Stealth[length=2.4mm]}, thick},
  lab/.style={font=\footnotesize, fill=white, inner sep=1.5pt, align=center}]
\node[sen] (wear) at (0,0.95) {\textbf{Wearable sensors}\\ ECG, temperature, fall sensor};
\node[sen] (amb) at (0,-0.95) {\textbf{Ambient sensors}\\ camera, microphone, motion sensor};
\node[grp, fit=(wear)(amb), label={[font=\small\bfseries]above:Patient environment}] (env) {};
\node[loc, right=14mm of env] (loc) {\textbf{Local processing unit}\\ physiological monitoring;\\
  activity recognition;\\ detection of falls and signs of distress};
\node[srv, right=20mm of loc] (srv) {\textbf{Central server}\\ electronic health record, predictive
  analytics, long-term storage};
\node[res, anchor=north west] (alert) at ($(loc.south)+(-0.3,-1.2)$) {\textbf{Alerts and insights}\\ clinician and caregiver dashboard,
  mobile application};
\node[res, text width=36mm, anchor=north] (dev) at ($(env.south)+(0,-0.75)$) {\textbf{Assistive devices}\\ local actuators
  that respond to the patient};
\draw[arr] (wear.east) -- (loc.west |- wear.east);
\draw[arr] (amb.east) -- (loc.west |- amb.east);
\node[lab] at ($(env.east)!0.5!(loc.west)$) {signals};
\draw[arr] (loc) -- node[lab, above=1mm] {events and\\ summaries} (srv);
\draw[arr] (loc.south -| alert.north) -- (alert.north);
\draw[arr] (srv.south) |- node[lab, pos=0.25, right=1mm] {insights} (alert.east);
\draw[arr] ($(loc.south west)+(0.5,0)$) |- (dev.east);
\end{tikzpicture}
